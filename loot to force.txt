-- ฟังก์ชันขายเฉพาะอาวุธ/เกราะขยะ โดยไม่แตะต้อง Ore, Material, EnchStone 100%
local function SafeSellNonEquipped()
    local count = 0
    pcall(function()
        local getDataRF = GetR("Backpack", "GetDataRF")
        local sellItemRE = GetR("Backpack", "TrySellItemRE")
        if not getDataRF or not sellItemRE then return end

        local data = getDataRF:InvokeServer()
        if not data or not data.have then return end

        local equiped = data.equiped or {}
        local equipMap = {}
        for _, eqUUID in pairs(equiped) do
            equipMap[eqUUID] = true
        end

        for uuid, item in pairs(data.have) do
            -- ป้องกัน Ore, Material, EnchStone ไม่ขายเด็ดขาด!
            if item.Type ~= "Ore" and item.Type ~= "Material" and item.Type ~= "EnchStone" then
                if not equipMap[uuid] and not item.Lock then
                    sellItemRE:FireServer(uuid, item.Number or 1)
                    count = count + 1
                end
            end
        end
    end)
    return count
end

-- ตรวจสอบและดึงช่วงชั้นล่าสุด / ชั้นที่ผู้เล่นเลือกบนหน้าจอ (เช่น 46~50)
local function GetTargetDungeonBracket()
    local targetStart = 46
    local targetEnd = 50

    pcall(function()
        for _, desc in ipairs(PlayerGui:GetDescendants()) do
            if desc:IsA("GuiButton") or desc:IsA("TextLabel") then
                local txt = desc.Text
                if txt and txt ~= "" then
                    local s, e = txt:match("(%d+)%s*~%s*(%d+)")
                    if s and e then
                        local sNum = tonumber(s)
                        local eNum = tonumber(e)
                        local isLocked = false
                        local parent = desc.Parent
                        if parent then
                            if parent:FindFirstChild("Lock") or parent:FindFirstChild("lock") or parent:FindFirstChild("LockIcon") then
                                isLocked = true
                            end
                        end
                        -- ถ้าเป็นปุ่มที่ปลดล็อคแล้ว และเป็นชั้นที่สูงที่สุดที่พบ
                        if not isLocked and sNum and sNum >= targetStart then
                            targetStart = sNum
                            targetEnd = eNum
                        end
                    end
                end
            end
        end
    end)

    return {
        StartRound = targetStart,
        EndRound = targetEnd,
        BracketStr = string.format("%d~%d", targetStart, targetEnd)
    }
end

-- สังหารและรัวดาเมจ 9.99e18 ให้ศัตรูทั้งหมดในดันเจี้ยนทันที
local function ClearAllDungeonEnemiesInstant()
    local killRE = GetR("Attack", "KillEnemyRE")
    local atkServiceRE = GetR("Attack", "AttackEnemyServiceRE")
    local targets = {}

    local folders = {
        workspace:FindFirstChild("EnemyFolder"),
        workspace:FindFirstChild("EnemyFolder_Server"),
        workspace:FindFirstChild("Dungeon") and workspace.Dungeon:FindFirstChild("EnemyFolder"),
        workspace:FindFirstChild("Dungeon") and workspace.Dungeon:FindFirstChild("Enemies"),
        workspace:FindFirstChild("DungeonEnemies"),
    }

    for _, f in ipairs(folders) do
        if f then
            for _, b in ipairs(f:GetChildren()) do
                if not b:GetAttribute("Dead") then
                    table.insert(targets, b.Name)
                    local hum = b:FindFirstChildOfClass("Humanoid")
                    if hum and hum.Health > 0 then
                        pcall(function() hum.Health = 0 end)
                    end
                end
            end
        end
    end

    for _, uuid in ipairs(targets) do
        if killRE then killRE:FireServer(uuid) end
        if atkServiceRE then
            for _ = 1, 5 do
                atkServiceRE:FireServer(
                    {uuid},
                    {Damage = 9.99e18, SkillID = "Weapon_ATK", Attacker = LocalPlayer},
                    workspace:GetServerTimeNow()
                )
            end
        end
    end

    return #targets
end

    startRound = startRound or 46
    endRound = endRound or 50
    bracketStr = bracketStr or string.format("%d~%d", startRound, endRound)

        UpdateDungeonStatus(txt)
        if statusCallback then statusCallback(txt) end
    end

    local success = false
    pcall(function()
        local dunFolder = ReplicatedStorage:FindFirstChild("Remote") and ReplicatedStorage.Remote:FindFirstChild("Dungeon")
        if not dunFolder then
            update("❌ ไม่พบ Remote Dungeon!")
            isDungeonRunning = false

            -- 2.1 คลิกเลือกปุ่มช่วงชั้นใน UI (เช่น ปุ่ม "46~50")

                for _, desc in ipairs(PlayerGui:GetDescendants()) do

                        local lbl = desc:FindFirstChildOfClass("TextLabel") or (desc:IsA("TextButton") and desc)
                        if lbl and lbl.Text and lbl.Text:find(bracketStr) then
                            local btn = desc:IsA("GuiButton") and desc or desc:FindFirstChildOfClass("GuiButton")
                            if btn then
                                if firesignal then
                                    pcall(function() firesignal(btn.MouseButton1Click) end)

                    end
                end
            end)
            task.wait(0.25)

            pcall(function()
                for _, desc in ipairs(PlayerGui:GetDescendants()) do
                    if desc:IsA("GuiButton") then
                        local txt = (desc:IsA("TextButton") and desc.Text) or ""
                        local childTxt = desc:FindFirstChildOfClass("TextLabel") and desc:FindFirstChildOfClass("TextLabel").Text or ""
                        if txt:find("เริ่ม") or childTxt:find("เริ่ม") or txt:find("Start") or childTxt:find("Start") then
                            if firesignal then
                                pcall(function() firesignal(desc.MouseButton1Click) end)
                                pcall(function() firesignal(desc.Activated) end)
                            end
                            pcall(function()
                                local VIM = game:GetService("VirtualInputManager")
                                local p = desc.AbsolutePosition
                                local s = desc.AbsoluteSize
                                VIM:SendMouseButtonEvent(p.X + s.X/2, p.Y + s.Y/2, 0, true, game, 0)

                            break
                        end
                    end
                end
            end)

            -- 2.3 ยิง Remote TryIntoDungeonRF สำรอง
            if intoRF then
                local tier = math.floor(endRound / 5)
                pcall(function() intoRF:InvokeServer(tier) end)
                pcall(function() intoRF:InvokeServer(startRound) end)
                pcall(function() intoRF:InvokeServer() end)
            end

            -- รอตรวจสอบการเข้าสู่ Dungeon
            local inFight = false
            for _ = 1, 16 do
                if LocalPlayer:GetAttribute("IntoFight") == "Dungeon" then
                    inFight = true
                    break
                end
                task.wait(0.25)
            end

            if not inFight then
                update(string.format("⚠️ เข้าดันเจี้ยน %s ไม่สำเร็จ (ตั๋วอาจหมด)", bracketStr))
                isDungeonRunning = false
                return
            end
        end

        update(string.format("⚔️ เข้าสู่ดันเจี้ยน %s สำเร็จ! กำลังเคลียร์ทุกรอบ...", bracketStr))

        -- 3. ลูปเอาชนะรอบทั้งหมด (เช่น 46, 47, 48, 49, 50)
        for curRound = startRound, endRound do
            update(string.format("🔥 กำลังล้างบางรอบที่ %d / %d...", curRound, endRound))

            -- 3.1 สั่งเริ่มรอบ
            if startRE then
                startRE:FireServer(curRound)
            end
            task.wait(0.2)

            -- 3.2 ลูปสังหารศัตรู/บอสในรอบด้วยดาเมจ 9.99e18 จนกว่าห้องจะสะอาด 100%
            local loopStart = tick()
            while tick() - loopStart < 12 do
                -- Auto God Mode ป้องกันผู้เล่นตายขณะสู้
                pcall(function()
                    local char = LocalPlayer.Character
                    if char then
                        local hum = char:FindFirstChildOfClass("Humanoid")
                        if hum then hum.Health = hum.MaxHealth end
                    end
                    LocalPlayer:SetAttribute("Dead", false)
                end)

                -- ฟาดศัตรูทุกตัวในห้อง
                local killed = ClearAllDungeonEnemiesInstant()

                -- ดูดแร่และรางวัล
                CollectDungeonLoot()

                -- ส่งคำสั่งผ่านรอบ
                if compRF then
                    pcall(function() compRF:InvokeServer(curRound) end)
                end

                -- ถ้าไม่มีศัตรูเหลือ และรันไปแล้วอย่างน้อย 0.4s
                if killed == 0 and (tick() - loopStart > 0.4) then
                    break
                end

                task.wait(0.18)
            end

            update(string.format("✅ ชนะรอบที่ %d สำเร็จ!", curRound))
            task.wait(0.25)
        end

        -- 4. ดูดแร่และของรางวัลรอบสุดท้าย
        CollectDungeonLoot()
        update(string.format("🏆 เอาชนะดันเจี้ยนชั้น %s สำเร็จเรียบร้อย 100%!", bracketStr))
        success = true
    end)

    isDungeonRunning = false
    return success
end
